animation.c

#include "animation.h"

#include <string.h>

void animation_reset(TileAnimation *animation)
{
    if (animation) memset(animation, 0, sizeof(*animation));
}

void animation_start(TileAnimation *animation, const MoveResult *move, float duration)
{
    if (!animation || !move) return;
    animation->active = true;
    animation->tile_value = move->tile_value;
    animation->from_index = move->from_index;
    animation->to_index = move->to_index;
    animation->elapsed = 0.0f;
    animation->duration = duration > 0.0f ? duration : 0.001f;
}

void animation_update(TileAnimation *animation, float frame_time)
{
    if (!animation || !animation->active || frame_time <= 0.0f) return;
    animation->elapsed += frame_time;
    if (animation->elapsed >= animation->duration) {
        animation->elapsed = animation->duration;
        animation->active = false;
    }
}

float animation_progress(const TileAnimation *animation)
{
    float t;
    if (!animation) return 1.0f;
    if (!animation->active) return 1.0f;
    if (animation->duration <= 0.0f) return 1.0f;

    t = animation->elapsed / animation->duration;
    if (t < 0.0f) t = 0.0f;
    if (t > 1.0f) t = 1.0f;

    /* Smoothstep easing: zero velocity at both ends of the visual transition. */
    return t * t * (3.0f - 2.0f * t);
}

#include "app.h"

static Direction input_direction(const GameInput *input)
{
    if (input->up) return DIR_UP;
    if (input->down) return DIR_DOWN;
    if (input->left) return DIR_LEFT;
    if (input->right) return DIR_RIGHT;
    return DIR_NONE;
}

static void set_status(App *app, const char *message)
{
    if (!app || !message) return;
    (void)snprintf(app->status, sizeof(app->status), "%s", message);
}

static void start_match(App *app)
{
    uint64_t seed = rng_splitmix64(app->base_seed + app->match_counter++);
    event_queue_clear(&app->events);
    animation_reset(&app->animation);
    sim_begin_match(&app->sim, seed, &app->events);
    app->mode = GAME_PLAYING;
    (void)snprintf(app->status, sizeof(app->status), "Puzzle started (seed %016llx)",
                   (unsigned long long)seed);
    LOG_GAME("match started; seed=%016llx", (unsigned long long)seed);
}

void app_init(App *app, uint64_t base_seed)
{
    if (!app) return;
    memset(app, 0, sizeof(*app));
    app->mode = GAME_TITLE;
    app->base_seed = base_seed;
    event_queue_clear(&app->events);
    animation_reset(&app->animation);
    set_status(app, "Press Enter or Space to start");
}

void app_shutdown(App *app)
{
    if (!app) return;
    sim_shutdown(&app->sim);
    animation_reset(&app->animation);
}

void app_frame(App *app, const GameInput *input, float frame_time)
{
    Direction direction;
    MoveResult move;

    if (!app || !input) return;

    if ((app->mode == GAME_TITLE || app->mode == GAME_OVER) && input->start) {
        start_match(app);
        return;
    }

    if (app->mode == GAME_PLAYING && input->pause) {
        app->mode = GAME_PAUSED;
        set_status(app, "Paused");
        return;
    }

    if (app->mode == GAME_PAUSED) {
        if (input->pause) {
            app->mode = GAME_PLAYING;
            set_status(app, "Resumed");
        }
        return;
    }

    if (app->mode != GAME_PLAYING) return;

    if (frame_time < 0.0f) frame_time = 0.0f;
    if (frame_time > MAX_FRAME_TIME) frame_time = MAX_FRAME_TIME;
    animation_update(&app->animation, frame_time);

    /* The simulation accepts discrete moves immediately. The application
       serializes input while the visual transition is in progress. */
    if (app->animation.active) return;

    if (app->sim.match_over) {
        app->mode = GAME_OVER;
        set_status(app, "Puzzle solved");
        LOG_GAME("match completed in %u moves", app->sim.move_count);
        return;
    }

    direction = input_direction(input);
    if (direction == DIR_NONE) return;

    if (sim_move(&app->sim, direction, &move, &app->events)) {
        animation_start(&app->animation, &move, TILE_ANIMATION_DURATION);
        (void)snprintf(app->status, sizeof(app->status), "Moved tile %u", move.tile_value);
        LOG_SIMULATION("tile=%u from=%u to=%u moves=%u",
                       move.tile_value, move.from_index, move.to_index,
                       app->sim.move_count);
    }
}

AppView app_view(const App *app)
{
    AppView view = {0};
    if (!app) return view;
    view.mode = app->mode;
    view.sim = &app->sim;
    view.animation = &app->animation;
    view.status = app->status;
    return view;
}

    }
    if (event_queue_recent(events, 0, &latest))
        DrawText(TextFormat("Event: %s", event_type_name(latest.type)), 20, 101, 13, SKYBLUE);
}

#ifndef EVENTS_H
#define EVENTS_H

#include <stdbool.h>
#include <stdint.h>

#define EVENT_RING_CAPACITY 256u

#if (EVENT_RING_CAPACITY & (EVENT_RING_CAPACITY - 1u)) != 0
#error "EVENT_RING_CAPACITY must be a power of two"
#endif

typedef enum {
    EVT_NONE = 0,
    EVT_MATCH_STARTED,
    EVT_TILE_SLID,
    EVT_MATCH_ENDED
} EventType;

typedef struct {
    uint32_t move_count;
    EventType type;
    uint8_t tile_value;
    uint8_t from_index;
    uint8_t to_index;
} SimEvent;

typedef struct {
    SimEvent events[EVENT_RING_CAPACITY];
    uint64_t head;
} EventQueue;

typedef struct {
    uint64_t next;
    uint32_t lost;
} EventCursor;

void event_queue_clear(EventQueue *queue);
void event_queue_push(EventQueue *queue, const SimEvent *event);
uint64_t event_queue_total(const EventQueue *queue);
bool event_queue_recent(const EventQueue *queue, uint32_t back, SimEvent *out);
void event_cursor_sync(const EventQueue *queue, EventCursor *cursor);
bool event_cursor_read(const EventQueue *queue, EventCursor *cursor, SimEvent *out);
const char *event_type_name(EventType type);

#endif


events.c

#include "events.h"

#define RING_MASK (EVENT_RING_CAPACITY - 1u)

void event_queue_clear(EventQueue *queue)
{
    if (queue) queue->head = 0;
}

void event_queue_push(EventQueue *queue, const SimEvent *event)
{
    if (!queue || !event) return;
    queue->events[queue->head & RING_MASK] = *event;
    queue->head++;
}

uint64_t event_queue_total(const EventQueue *queue)
{
    return queue ? queue->head : 0;
}

bool event_queue_recent(const EventQueue *queue, uint32_t back, SimEvent *out)
{
    uint64_t available;
    if (!queue || !out) return false;
    available = queue->head < EVENT_RING_CAPACITY ? queue->head : EVENT_RING_CAPACITY;
    if ((uint64_t)back >= available) return false;
    *out = queue->events[(queue->head - 1u - back) & RING_MASK];
    return true;
}

void event_cursor_sync(const EventQueue *queue, EventCursor *cursor)
{
    if (!cursor) return;
    cursor->next = queue ? queue->head : 0;
    cursor->lost = 0;
}

bool event_cursor_read(const EventQueue *queue, EventCursor *cursor, SimEvent *out)
{
    uint64_t oldest;
    if (!queue || !cursor || !out) return false;
    if (cursor->next > queue->head) cursor->next = queue->head;
    oldest = queue->head > EVENT_RING_CAPACITY ? queue->head - EVENT_RING_CAPACITY : 0;
    if (cursor->next < oldest) {
        cursor->lost += (uint32_t)(oldest - cursor->next);
        cursor->next = oldest;
    }
    if (cursor->next >= queue->head) return false;
    *out = queue->events[cursor->next & RING_MASK];
    cursor->next++;
    return true;
}

const char *event_type_name(EventType type)
{
    switch (type) {
        case EVT_MATCH_STARTED: return "MATCH_START";
        case EVT_TILE_SLID:     return "TILE_SLID";
        case EVT_MATCH_ENDED:   return "MATCH_END";
        default:                return "NONE";
    }
}

log.c

#include "log.h"

#include <stdarg.h>
#include <stdio.h>

static uint32_t enabled_categories;

#define LOG_CATEGORY_NAME(category, name) [category] = name,
static const char *log_category_names[LOG_COUNT] = {
    LOG_CATEGORIES(LOG_CATEGORY_NAME)
};
#undef LOG_CATEGORY_NAME

void log_init(void) { enabled_categories = 0; }

void log_enable(LogCategory category)
{
    if ((unsigned)category < LOG_COUNT && (unsigned)category < 32u)
        enabled_categories |= (1u << (unsigned)category);
}

void log_disable(LogCategory category)
{
    if ((unsigned)category < LOG_COUNT && (unsigned)category < 32u)
        enabled_categories &= ~(1u << (unsigned)category);
}

int log_is_enabled(LogCategory category)
{
    if ((unsigned)category >= LOG_COUNT || (unsigned)category >= 32u) return 0;
    return (enabled_categories & (1u << (unsigned)category)) != 0;
}

void log_message(LogCategory category, const char *file, int line,
                 const char *format, ...)
{
    va_list args;
    if ((unsigned)category >= LOG_COUNT || !log_is_enabled(category)) return;

    printf("[%s] %s:%d: ", log_category_names[category], file, line);
    va_start(args, format);
    vprintf(format, args);
    va_end(args);
    putchar('\n');
    fflush(stdout);
}

#include "app.h"
#include "config.h"
#include "debug.h"
#include "input.h"
#include "log.h"
#include "raylib.h"
#include "render.h"

#include <time.h>

int main(void)
{
    App app;
    DebugState debug;

    InitWindow(SCREEN_WIDTH, SCREEN_HEIGHT, "15-Puzzle - Discrete Logic");
    SetExitKey(KEY_NULL);
    SetTargetFPS(TARGET_FPS);

    log_init();
    log_enable(PUZZLE_LOG_GAME);
    log_enable(PUZZLE_LOG_SIMULATION);
    debug_init(&debug);
    app_init(&app, (uint64_t)time(NULL));

    while (!WindowShouldClose()) {
        GameInput input = {0};
        float frame_time = GetFrameTime();
        AppView view;

        input_sample(&input);
        if (input.debug_toggle) debug.enabled = !debug.enabled;
        app_frame(&app, &input, frame_time);
        view = app_view(&app);
        debug_update(&debug, frame_time);

        BeginDrawing();
            render_frame(&view);
            debug_draw(&debug, &view, &app.events);
        EndDrawing();
    }

    app_shutdown(&app);
    CloseWindow();
    return 0;
}

#include "render.h"

#define TILE_SIZE 88
#define TILE_GAP 8
#define BOARD_PIXEL_SIZE (BOARD_SIZE * (TILE_SIZE + TILE_GAP) - TILE_GAP)
#define BOARD_OFFSET_X ((SCREEN_WIDTH - BOARD_PIXEL_SIZE) / 2)
#define BOARD_OFFSET_Y ((SCREEN_HEIGHT - BOARD_PIXEL_SIZE) / 2 + 18)

static const Color COLOR_BACKGROUND = { 18, 20, 27, 255 };
static const Color COLOR_PANEL = { 32, 36, 47, 255 };
static const Color COLOR_TILE = { 225, 229, 235, 255 };
static const Color COLOR_TILE_EDGE = { 150, 160, 175, 255 };
static const Color COLOR_TEXT = { 235, 238, 244, 255 };

static float lerp(float a, float b, float t)
{
    return a + (b - a) * t;
}

static void draw_centered(const char *text, int y, int font_size, Color color)
{
    int width = MeasureText(text, font_size);
    DrawText(text, (SCREEN_WIDTH - width) / 2, y, font_size, color);
}

static void draw_title(void)
{
    draw_centered("15-PUZZLE", 165, 48, COLOR_TEXT);
    draw_centered("DISCRETE LOGIC / VISUAL ANIMATION", 226, 18, LIGHTGRAY);
    draw_centered("PRESS ENTER OR SPACE", 310, 22, YELLOW);
    draw_centered("Arrow keys or WASD to slide tiles", 356, 18, COLOR_TEXT);
}

static void draw_tile(uint8_t value, float row, float col)
{
    int x = BOARD_OFFSET_X + (int)(col * (float)(TILE_SIZE + TILE_GAP));
    int y = BOARD_OFFSET_Y + (int)(row * (float)(TILE_SIZE + TILE_GAP));
    const char *label = TextFormat("%u", (unsigned)value);
    int font_size = 30;
    int text_width = MeasureText(label, font_size);

    DrawRectangle(x + 2, y + 3, TILE_SIZE, TILE_SIZE, Fade(BLACK, 0.28f));
    DrawRectangle(x, y, TILE_SIZE, TILE_SIZE, COLOR_TILE);
    DrawRectangleLines(x, y, TILE_SIZE, TILE_SIZE, COLOR_TILE_EDGE);
    DrawText(label, x + (TILE_SIZE - text_width) / 2,
             y + (TILE_SIZE - font_size) / 2, font_size, COLOR_PANEL);
}

static void draw_board(const AppView *view)
{
    const SimulationState *sim = view->sim;
    const TileAnimation *animation = view->animation;
    int i;

    DrawRectangle(BOARD_OFFSET_X - 14, BOARD_OFFSET_Y - 14,
                  BOARD_PIXEL_SIZE + 28, BOARD_PIXEL_SIZE + 28, COLOR_PANEL);
    DrawRectangleLines(BOARD_OFFSET_X - 14, BOARD_OFFSET_Y - 14,
                       BOARD_PIXEL_SIZE + 28, BOARD_PIXEL_SIZE + 28, GRAY);

    for (i = 0; i < TOTAL_TILES; ++i) {
        int row = i / BOARD_SIZE;
        int col = i % BOARD_SIZE;
        int x = BOARD_OFFSET_X + col * (TILE_SIZE + TILE_GAP);
        int y = BOARD_OFFSET_Y + row * (TILE_SIZE + TILE_GAP);
        DrawRectangle(x, y, TILE_SIZE, TILE_SIZE, Fade(BLACK, 0.35f));
    }

    for (i = 0; i < TOTAL_TILES; ++i) {
        uint8_t value = sim->board[i];
        if (value == 0) continue;
        if (animation && animation->active && value == animation->tile_value) continue;
        draw_tile(value, (float)(i / BOARD_SIZE), (float)(i % BOARD_SIZE));
    }

    if (animation && animation->active) {
        float t = animation_progress(animation);
        float from_row = (float)(animation->from_index / BOARD_SIZE);
        float from_col = (float)(animation->from_index % BOARD_SIZE);
        float to_row = (float)(animation->to_index / BOARD_SIZE);
        float to_col = (float)(animation->to_index % BOARD_SIZE);
        draw_tile(animation->tile_value,
                  lerp(from_row, to_row, t),
                  lerp(from_col, to_col, t));
    }

    DrawText(TextFormat("MOVES  %u", sim->move_count), BOARD_OFFSET_X,
             BOARD_OFFSET_Y - 43, 20, COLOR_TEXT);
    DrawText("ARROWS / WASD: MOVE     P / ESC: PAUSE     F3: DEBUG",
             BOARD_OFFSET_X, BOARD_OFFSET_Y + BOARD_PIXEL_SIZE + 29, 14, LIGHTGRAY);
}

static void draw_pause(void)
{
    DrawRectangle(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT, Fade(BLACK, 0.68f));
    draw_centered("PAUSED", SCREEN_HEIGHT / 2 - 16, 34, COLOR_TEXT);
    draw_centered("PRESS P OR ESC TO RESUME", SCREEN_HEIGHT / 2 + 28, 17, YELLOW);
}

static void draw_game_over(const AppView *view)
{
    DrawRectangle(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT, Fade(BLACK, 0.72f));
    draw_centered("PUZZLE SOLVED!", SCREEN_HEIGHT / 2 - 54, 38, GREEN);
    draw_centered(TextFormat("TOTAL MOVES: %u", view->sim->move_count),
                  SCREEN_HEIGHT / 2 + 2, 22, COLOR_TEXT);
    draw_centered("PRESS ENTER TO PLAY AGAIN", SCREEN_HEIGHT / 2 + 52, 18, YELLOW);
}

void render_frame(const AppView *view)
{
    if (!view) return;
    ClearBackground(COLOR_BACKGROUND);

    switch (view->mode) {
        case GAME_TITLE:
            draw_title();
            break;
        case GAME_PLAYING:
            draw_board(view);
            break;
        case GAME_PAUSED:
            draw_board(view);
            draw_pause();
            break;
        case GAME_OVER:
            draw_board(view);
            draw_game_over(view);
            break;
        default:
            draw_centered("INVALID GAME STATE", SCREEN_HEIGHT / 2, 24, RED);
            break;
    }
}

rng.c

#include "rng.h"

uint64_t rng_splitmix64(uint64_t x)
{
    x += 0x9e3779b97f4a7c15ULL;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
    return x ^ (x >> 31);
}

void rng_seed(Rng *rng, uint64_t seed, uint64_t stream)
{
    uint64_t initial = rng_splitmix64(seed + stream * 0x9e3779b97f4a7c15ULL);
    rng->state = initial == 0ULL ? 0x123456789ABCDEFULL : initial;
}

uint64_t rng_next_u64(Rng *rng)
{
    uint64_t x = rng->state;
    x ^= x >> 12;
    x ^= x << 25;
    x ^= x >> 27;
    rng->state = x;
    return x * 0x2545f4914f6cdd1dULL;
}

uint32_t rng_range(Rng *rng, uint32_t min, uint32_t max)
{
    if (min >= max) return min;
    return min + (uint32_t)(rng_next_u64(rng) % ((uint64_t)max - min + 1ULL));
}

static bool direction_delta(Direction direction, int *dr, int *dc)
{
    switch (direction) {
        case DIR_UP:    *dr = -1; *dc =  0; return true;
        case DIR_DOWN:  *dr =  1; *dc =  0; return true;
        case DIR_LEFT:  *dr =  0; *dc = -1; return true;
        case DIR_RIGHT: *dr =  0; *dc =  1; return true;
        default: return false;
    }
}

static Direction opposite_direction(Direction direction)
{
    switch (direction) {
        case DIR_UP:    return DIR_DOWN;
        case DIR_DOWN:  return DIR_UP;
        case DIR_LEFT:  return DIR_RIGHT;
        case DIR_RIGHT: return DIR_LEFT;
        default:        return DIR_NONE;
    }
}

/* Direction describes the direction in which the tile moves. */
static bool find_source_index(const SimulationState *state, Direction direction,
                              uint8_t *source_index)
{
    int dr = 0;
    int dc = 0;
    int empty_row;
    int empty_col;
    int source_row;
    int source_col;

    if (!state || !source_index || !direction_delta(direction, &dr, &dc))
        return false;

    empty_row = state->empty_index / BOARD_SIZE;
    empty_col = state->empty_index % BOARD_SIZE;
    source_row = empty_row - dr;
    source_col = empty_col - dc;

    if (source_row < 0 || source_row >= BOARD_SIZE ||
        source_col < 0 || source_col >= BOARD_SIZE)
        return false;

    *source_index = (uint8_t)(source_row * BOARD_SIZE + source_col);
    return state->board[*source_index] != 0;
}

bool sim_can_move(const SimulationState *state, Direction direction)
{
    uint8_t source_index;
    if (!state || state->match_over || state->is_solved) return false;
    return find_source_index(state, direction, &source_index);
}

/* This is the only operation that changes the board layout. No time or position data. */
static bool apply_swap(SimulationState *state, Direction direction, MoveResult *result)
{
    uint8_t source_index;
    uint8_t destination_index;
    uint8_t tile_value;

    if (!find_source_index(state, direction, &source_index)) return false;

    destination_index = state->empty_index;
    tile_value = state->board[source_index];
    assert(tile_value >= 1 && tile_value < TOTAL_TILES);

    state->board[destination_index] = tile_value;
    state->board[source_index] = 0;
    state->empty_index = source_index;

    if (result) {
        result->tile_value = tile_value;
        result->from_index = source_index;
        result->to_index = destination_index;
    }
    return true;
}

bool sim_check_solved(const SimulationState *state)
{
    int i;
    if (!state) return false;
    for (i = 0; i < TOTAL_TILES - 1; ++i) {
        if (state->board[i] != (uint8_t)(i + 1)) return false;
    }
    return state->board[TOTAL_TILES - 1] == 0;
}

bool sim_move(SimulationState *state, Direction direction, MoveResult *result,
              EventQueue *events)
{
    MoveResult move;
    SimEvent event;

    if (!state || state->match_over || state->is_solved) return false;
    if (!apply_swap(state, direction, &move)) return false;

    state->move_count++;
    state->is_solved = sim_check_solved(state);
    state->match_over = state->is_solved;

    if (result) *result = move;

    event = (SimEvent){
        .move_count = state->move_count,
        .type = EVT_TILE_SLID,
        .tile_value = move.tile_value,
        .from_index = move.from_index,
        .to_index = move.to_index
    };
    event_queue_push(events, &event);

    if (state->match_over) {
        event.type = EVT_MATCH_ENDED;
        event_queue_push(events, &event);
    }
    return true;
}

static void shuffle_board(SimulationState *state)
{
    Direction directions[4] = { DIR_UP, DIR_DOWN, DIR_LEFT, DIR_RIGHT };
    Direction previous = DIR_NONE;
    int step;

    for (step = 0; step < SHUFFLE_MOVE_COUNT; ++step) {
        Direction candidates[4];
        int candidate_count = 0;
        int i;

        for (i = 0; i < 4; ++i) {
            Direction direction = directions[i];
            uint8_t source_index;
            if (!find_source_index(state, direction, &source_index)) continue;
            if (direction == opposite_direction(previous)) continue;
            candidates[candidate_count++] = direction;
        }

        /* At a corner/edge there is always a legal move. If excluding the
           reverse direction removed all options, allow it rather than stall. */
        if (candidate_count == 0) {
            for (i = 0; i < 4; ++i) {
                uint8_t source_index;
                if (find_source_index(state, directions[i], &source_index))
                    candidates[candidate_count++] = directions[i];
            }
        }

        if (candidate_count > 0) {
            Direction chosen = candidates[rng_range(&state->rng, 0, (uint32_t)candidate_count - 1u)];
            (void)apply_swap(state, chosen, NULL);
            previous = chosen;
        }
    }
}

void sim_begin_match(SimulationState *state, uint64_t seed, EventQueue *events)
{
    int i;
    if (!state) return;

    memset(state, 0, sizeof(*state));
    state->seed = seed;
    rng_seed(&state->rng, seed, 1);

    for (i = 0; i < TOTAL_TILES - 1; ++i)
        state->board[i] = (uint8_t)(i + 1);
    state->board[TOTAL_TILES - 1] = 0;
    state->empty_index = TOTAL_TILES - 1;

    shuffle_board(state);

    /* A random walk is solvable by construction. Keep the explicit check as
       a guard against future changes to the shuffle implementation. */
    if (sim_check_solved(state)) {
        for (i = 0; i < 4; ++i) {
            uint8_t source_index;
            if (find_source_index(state, (Direction)(i + 1), &source_index)) {
                (void)apply_swap(state, (Direction)(i + 1), NULL);
                break;
            }
        }
    }

    state->move_count = 0;
    state->is_solved = false;
    state->match_over = false;

    {
        SimEvent event = { .move_count = 0, .type = EVT_MATCH_STARTED };
        event_queue_push(events, &event);
    }
}

void sim_shutdown(SimulationState *state)
{
    if (state) memset(state, 0, sizeof(*state));
}
