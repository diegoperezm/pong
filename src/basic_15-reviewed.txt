#include "app.h"

#include <stdarg.h>

#if defined(__GNUC__) || defined(__clang__)
__attribute__((format(printf, 2, 3)))
#endif
static void set_status(App *app, const char *fmt, ...) {
    va_list args;
    va_start(args, fmt);
    vsnprintf(app->status, sizeof app->status, fmt, args);
    va_end(args);
}

static void make_snapshot(const App *app, RenderSnapshot *snap) {
    const SimulationState *s = &app->sim;
    snap->mode = app->mode;
    snap->move_count = s->move_count;
    snap->game_time = (float)s->tick * SIM_DT;
    snap->is_solved = s->is_solved;

    for (int i = 0; i < TOTAL_TILES - 1; ++i) {
        snap->tiles[i].value = s->tiles[i].value;
        snap->tiles[i].row   = s->tiles[i].position.row;
        snap->tiles[i].col   = s->tiles[i].position.col;
    }
}

static void snapshots_reset(App *app) {
    make_snapshot(app, &app->current);
    app->previous = app->current;
}

static void begin_match_common(App *app, uint64_t seed) {
    event_queue_clear(&app->events);
    sim_begin_match(&app->sim, seed, &app->events);
    app->accumulator = 0.0;
}

static void start_live_match(App *app) {
    uint64_t seed = rng_splitmix64(app->base_seed + app->match_counter++);
    input_source_use_live(&app->source);
    begin_match_common(app, seed);
    app->mode = GAME_PLAYING;
    snapshots_reset(app);
    set_status(app, "puzzle started (seed %016llx)", (unsigned long long)seed);
}

static void run_tick(App *app, bool snapshot) {
    SimulationState *sim = &app->sim;
    const uint32_t tick = sim->tick;
    TickInput in = input_source_at(&app->source, tick);

    if (snapshot) app->previous = app->current;
    sim_step(sim, &in, (float)SIM_DT, &app->events);
    input_source_consume_one_shot(&app->source);

    if (sim->match_over) {
        app->mode = GAME_OVER;
        app->accumulator = 0.0;
        // Fix Critical 1: Update snapshot after changing mode to GAME_OVER
        if (snapshot) {
            make_snapshot(app, &app->current);
            app->previous = app->current;
        }
    } else {
        if (snapshot) make_snapshot(app, &app->current);
    }
}

static void handle_input(App *app, const GameInput *in) {
    switch (app->mode) {
        case GAME_TITLE:
        case GAME_OVER:
            if (in->start) start_live_match(app);
            break;
        case GAME_PLAYING:
            if (in->pause) {
                app->mode = GAME_PAUSED;
                input_source_clear(&app->source); // Fix Medium 7: clear pending input on pause
            }
            break;
        case GAME_PAUSED:
            if (in->pause) {
                app->mode = GAME_PLAYING;
                input_source_clear(&app->source);
            }
            break;
    }
}

void app_init(App *app, uint64_t base_seed) {
    memset(app, 0, sizeof(*app));
    app->mode = GAME_TITLE;
    app->base_seed = base_seed;
    app->alpha = 1.0f;
    input_source_use_live(&app->source);
    event_queue_clear(&app->events);
    snapshots_reset(app);
}

void app_shutdown(App *app) {
    sim_shutdown(&app->sim);
}

void app_frame(App *app, const GameInput *input, double frame_time) {
    GameMode before = app->mode;
    handle_input(app, input);

    if (app->mode != before) {
        app->accumulator = 0.0;
        snapshots_reset(app);
    }

    if (app->mode == GAME_PLAYING) {
        Direction dir = DIR_NONE;
        if (input->up)    dir = DIR_UP;
        if (input->down)  dir = DIR_DOWN;
        if (input->left)  dir = DIR_LEFT;
        if (input->right) dir = DIR_RIGHT;

        if (dir != DIR_NONE) {
            input_source_set_pending(&app->source, (TickInput){ .move_dir = dir });
        }

        app->accumulator += frame_time;

        while (app->accumulator >= (double)SIM_DT) {
            run_tick(app, true);
            app->accumulator -= (double)SIM_DT;

            if (app->mode != GAME_PLAYING) {
                app->accumulator = 0.0;
                break;
            }
        }

        if (app->mode == GAME_PLAYING) {
            app->alpha = (float)(app->accumulator / (double)SIM_DT);
        } else {
            app->alpha = 1.0f;
        }
    } else {
        app->alpha = 1.0f;
    }
}

AppView app_view(const App *app) {
    return (AppView){ &app->previous, &app->current, app->alpha };
}

#define EVENT_RING_CAPACITY 256u

_Static_assert((EVENT_RING_CAPACITY & (EVENT_RING_CAPACITY - 1u)) == 0u,
               "EVENT_RING_CAPACITY must be a power of two");

typedef enum {
    EVT_NONE = 0,
    EVT_MATCH_STARTED,
    EVT_TILE_SLID,
    EVT_TILE_SETTLED,
    EVT_MATCH_ENDED
} EventType;

typedef struct {
    uint32_t tick;
    EventType type;
    int tile_value;
    int score_player;
} SimEvent;

typedef struct {
    SimEvent events[EVENT_RING_CAPACITY];
    uint64_t head;
} EventQueue;

typedef struct {
    uint64_t next;
    uint32_t lost;
} EventCursor;

void event_queue_clear(EventQueue *q);
void event_queue_push(EventQueue *q, const SimEvent *e);

bool event_queue_recent(const EventQueue *q, uint32_t back, SimEvent *out);

events.c

#include "events.h"
#define RING_MASK (EVENT_RING_CAPACITY - 1u)

void event_queue_clear(EventQueue *q) { if(q) q->head = 0; }
void event_queue_push(EventQueue *q, const SimEvent *e) {
    if(!q || !e) return;
    q->events[q->head & RING_MASK] = *e;
    q->head++;
}
uint64_t event_queue_total(const EventQueue *q) { return q ? q->head : 0; }
bool event_queue_recent(const EventQueue *q, uint32_t back, SimEvent *out) {
    if(!q || !out) return false;
    uint64_t available = q->head < EVENT_RING_CAPACITY ? q->head : EVENT_RING_CAPACITY;
    if((uint64_t)back >= available) return false;
    *out = q->events[(q->head - 1u - back) & RING_MASK];
    return true;
}
void event_cursor_sync(const EventQueue *q, EventCursor *c) {
    if (!c) return;
    c->next = q ? q->head : 0;
    c->lost = 0;
}
bool event_cursor_read(const EventQueue *q, EventCursor *c, SimEvent *out) {
    if(!q || !c || !out) return false;
    if(c->next > q->head) c->next = q->head;
    uint64_t oldest = q->head > EVENT_RING_CAPACITY ? q->head - EVENT_RING_CAPACITY : 0;
    if(c->next < oldest) { c->lost += (uint32_t)(oldest - c->next); c->next = oldest; }
    if(c->next >= q->head) return false;
    *out = q->events[c->next & RING_MASK];
    c->next++;
    return true;
}
const char *event_type_name(EventType type) {
    switch(type) {
        case EVT_MATCH_STARTED: return "MATCH_START";
        case EVT_TILE_SLID:     return "TILE_SLID";
        case EVT_TILE_SETTLED:  return "TILE_SETTLED";
        case EVT_MATCH_ENDED:   return "MATCH_END";
        default: return "NONE";
    }
}

input_source.c

#include "input_source.h"

void input_source_use_live(InputSource *s) {
    if (!s) return;
    s->kind = INPUT_SOURCE_LIVE;
    s->pending_input = (TickInput){ DIR_NONE };
}

void input_source_set_pending(InputSource *s, TickInput held) {
    if (!s) return;
    s->pending_input = held;
}

void input_source_consume_one_shot(InputSource *s) {
    if (!s) return;
    s->pending_input.move_dir = DIR_NONE;
}

void input_source_clear(InputSource *s) {
    if (!s) return;
    s->pending_input = (TickInput){ DIR_NONE };
}

TickInput input_source_at(InputSource *s, uint32_t tick) {
    (void)tick;
    if (s && s->kind == INPUT_SOURCE_LIVE)
        return s->pending_input;
    return (TickInput){ DIR_NONE };
}

log.c

#include "log.h"
#include <stdio.h>
#include <stdarg.h>

static uint32_t enabled_categories;

#define LOG_CATEGORY_NAME(category, name) [category] = name,

static const char *log_category_names[LOG_COUNT] = {
    LOG_CATEGORIES(LOG_CATEGORY_NAME)
};

#undef LOG_CATEGORY_NAME

void log_init(void) { enabled_categories = 0; }
void log_enable(LogCategory c) { if ((unsigned)c < LOG_COUNT && c < 32) enabled_categories |= (1u << c); }
void log_disable(LogCategory c) { if ((unsigned)c < LOG_COUNT && c < 32) enabled_categories &= ~(1u << c); }

int log_is_enabled(LogCategory c) {
    if ((unsigned)c >= LOG_COUNT || c >= 32) return 0;
    return (enabled_categories & (1u << c)) != 0;
}

void log_message(LogCategory category, const char *file, int line, const char *format, ...) {
    if ((unsigned)category >= LOG_COUNT || !log_is_enabled(category)) return;
    
    printf("[%s] %s:%d: ", log_category_names[category], file, line);
    va_list args;
    va_start(args, format);
    vprintf(format, args);
    va_end(args);
    printf("\n");
    fflush(stdout);
}

#include <time.h>
#include "raylib.h"
#include "config.h"
#include "app.h"
#include "input.h"
#include "log.h"
#include "debug.h"
#include "render.h"

int main(void) {
    InitWindow(SCREEN_WIDTH, SCREEN_HEIGHT, "15-Puzzle Physics Simulation");
    SetExitKey(KEY_NULL);
    SetTargetFPS(60);

    log_init();
    log_enable(PUZZLE_LOG_GAME);
    log_enable(PUZZLE_LOG_SIMULATION);

    static App app;
    app_init(&app, (uint64_t)time(NULL));

    DebugState debug;
    debug_init(&debug);

    while (!WindowShouldClose()) {
        GameInput input;
        input_sample(&input);

        double raw_frame_time = GetFrameTime();
        double frame_time = raw_frame_time > MAX_FRAME_TIME ? MAX_FRAME_TIME : raw_frame_time;

        if (input.debug_toggle)
            debug.enabled = !debug.enabled;

        app_frame(&app, &input, frame_time);

        AppView view = app_view(&app);
        DebugHistory history = { .events = &app.events, .tick = app.sim.tick, .seed = app.sim.seed, .status = app.status };

        debug_update(&debug, view.current, (float)raw_frame_time);

        BeginDrawing();
            ClearBackground(BLACK);
            render_frame(view.previous, view.current, view.alpha);
            debug_draw(&debug, view.previous, view.current, view.alpha, &history);
        EndDrawing();
    }

    app_shutdown(&app);
    CloseWindow();
    return 0;
}

#include "render.h"

#define TILE_SIZE 90
#define TILE_GAP  6
#define BOARD_OFFSET_X ((SCREEN_WIDTH - (BOARD_SIZE * (TILE_SIZE + TILE_GAP) - TILE_GAP)) / 2)
#define BOARD_OFFSET_Y ((SCREEN_HEIGHT - (BOARD_SIZE * (TILE_SIZE + TILE_GAP) - TILE_GAP)) / 2 + 20)

void render_init(void) {}
void render_shutdown(void) {}

static float lerp(float a, float b, float t) {
    return a + (b - a) * t;
}

static void draw_title(void) {
    const char *title = "15-PUZZLE PHYSICS";
    const char *prompt = "PRESS ENTER OR SPACE";
    const char *instr = "Use Arrow Keys to Slide Tiles";

    DrawText(title, (SCREEN_WIDTH / 2) - (MeasureText(title, 36) / 2), 140, 36, RAYWHITE);
    DrawText(instr, (SCREEN_WIDTH / 2) - (MeasureText(instr, 20) / 2), 210, 20, LIGHTGRAY);
    DrawText(prompt, (SCREEN_WIDTH / 2) - (MeasureText(prompt, 20) / 2), 260, 20, YELLOW);
}

static void draw_board(const RenderSnapshot *previous, const RenderSnapshot *current, float alpha) {
    int board_w = BOARD_SIZE * (TILE_SIZE + TILE_GAP) - TILE_GAP;
    DrawRectangle(BOARD_OFFSET_X - 12, BOARD_OFFSET_Y - 12, board_w + 24, board_w + 24, DARKGRAY);
    DrawRectangleLines(BOARD_OFFSET_X - 12, BOARD_OFFSET_Y - 12, board_w + 24, board_w + 24, GRAY);

    for (int r = 0; r < BOARD_SIZE; ++r) {
        for (int c = 0; c < BOARD_SIZE; ++c) {
            int x = BOARD_OFFSET_X + c * (TILE_SIZE + TILE_GAP);
            int y = BOARD_OFFSET_Y + r * (TILE_SIZE + TILE_GAP);
            DrawRectangle(x, y, TILE_SIZE, TILE_SIZE, Fade(BLACK, 0.4f));
        }
    }

    for (int i = 0; i < TOTAL_TILES - 1; ++i) {
        uint8_t val = current->tiles[i].value;
        if (val == 0) continue;

        float row = lerp(previous->tiles[i].row, current->tiles[i].row, alpha);
        float col = lerp(previous->tiles[i].col, current->tiles[i].col, alpha);

        int x = BOARD_OFFSET_X + (int)(col * (float)(TILE_SIZE + TILE_GAP));
        int y = BOARD_OFFSET_Y + (int)(row * (float)(TILE_SIZE + TILE_GAP));

        DrawRectangle(x, y, TILE_SIZE, TILE_SIZE, LIGHTGRAY);
        DrawRectangleLines(x, y, TILE_SIZE, TILE_SIZE, DARKGRAY);

        const char *text = TextFormat("%d", val);
        int font_size = 32;
        int text_w = MeasureText(text, font_size);
        DrawText(text, x + (TILE_SIZE - text_w) / 2, y + (TILE_SIZE - font_size) / 2, font_size, BLACK);
    }

    DrawText(TextFormat("Moves: %u", current->move_count), BOARD_OFFSET_X, BOARD_OFFSET_Y - 40, 22, RAYWHITE);
    DrawText("P: Pause | F3: Debug", SCREEN_WIDTH - 180, BOARD_OFFSET_Y - 40, 16, LIGHTGRAY);
}

static void draw_pause(void) {
    DrawRectangle(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT, Fade(BLACK, 0.7f));
    const char *text = "PAUSED";
    DrawText(text, (SCREEN_WIDTH / 2) - (MeasureText(text, 30) / 2), SCREEN_HEIGHT / 2 - 20, 30, RAYWHITE);
}

static void draw_game_over(const RenderSnapshot *current) {
    DrawRectangle(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT, Fade(BLACK, 0.75f));
    const char *text = "PUZZLE SOLVED!";
    const char *score_text = TextFormat("Total Moves: %u", current->move_count);
    const char *prompt = "PRESS ENTER TO PLAY AGAIN";

    DrawText(text, (SCREEN_WIDTH / 2) - (MeasureText(text, 35) / 2), SCREEN_HEIGHT / 2 - 50, 35, GREEN);
    DrawText(score_text, (SCREEN_WIDTH / 2) - (MeasureText(score_text, 22) / 2), SCREEN_HEIGHT / 2, 22, RAYWHITE);
    DrawText(prompt, (SCREEN_WIDTH / 2) - (MeasureText(prompt, 20) / 2), SCREEN_HEIGHT / 2 + 50, 20, YELLOW);
}

void render_frame(const RenderSnapshot *previous, const RenderSnapshot *current, float alpha) {
    switch (current->mode) {
        case GAME_TITLE:
            draw_title();
            break;
        case GAME_PLAYING:
            draw_board(previous, current, alpha);
            break;
        case GAME_PAUSED:
            draw_board(previous, current, alpha);
            draw_pause();
            break;
        case GAME_OVER:
            draw_board(previous, current, alpha);
            draw_game_over(current);
            break;
    }
}

rng.c

#include "rng.h"

uint64_t rng_splitmix64(uint64_t x) {
    x += 0x9e3779b97f4a7c15ULL;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
    return x ^ (x >> 31);
}

void rng_seed(Rng *rng, uint64_t seed, uint64_t stream) {
    uint64_t initial = rng_splitmix64(seed + stream * 0x9e3779b97f4a7c15ULL);
    // Prevent all-zero state for xorshift
    rng->state = (initial == 0ULL) ? 0x123456789ABCDEFULL : initial;
}

uint64_t rng_next_u64(Rng *rng) {
    uint64_t x = rng->state;
    x ^= x >> 12;
    x ^= x << 25;
    x ^= x >> 27;
    rng->state = x;
    return x * 0x2545f4914f6cdd1dULL;
}

uint32_t rng_range(Rng *rng, uint32_t min, uint32_t max) {
    if (min >= max) return min;
    uint64_t span = (uint64_t)max - min + 1ULL;
    return min + (uint32_t)(rng_next_u64(rng) % span);
}

#include "simulation.h"

static bool check_solved(const SimulationState *sim) {
    uint8_t expected = 1;
    for (int r = 0; r < BOARD_SIZE; ++r) {
        for (int c = 0; c < BOARD_SIZE; ++c) {
            if (r == BOARD_SIZE - 1 && c == BOARD_SIZE - 1) {
                if (sim->logical_board[r][c] != 0) return false;
            } else {
                if (sim->logical_board[r][c] != expected++) return false;
            }
        }
    }
    return true;
}

static bool all_tiles_settled(const SimulationState *sim) {
    for (int i = 0; i < TOTAL_TILES - 1; ++i) {
        if (sim->tiles[i].is_moving)
            return false;
    }
    return true;
}

static bool any_tile_moving(const SimulationState *sim) {
    return !all_tiles_settled(sim);
}

static bool execute_slide(SimulationState *sim, int dr, int dc, int *moved_val) {
    // Physical movement policy: do not accept a new move while any tile is still sliding
    if (any_tile_moving(sim)) {
        return false;
    }

    int target_r = sim->empty_row + dr;
    int target_c = sim->empty_col + dc;

    if (target_r < 0 || target_r >= BOARD_SIZE || target_c < 0 || target_c >= BOARD_SIZE) {
        return false;
    }

    *moved_val = sim->logical_board[target_r][target_c];
    assert(*moved_val >= 1 && *moved_val < TOTAL_TILES);

    sim->logical_board[sim->empty_row][sim->empty_col] = *moved_val;
    sim->logical_board[target_r][target_c] = 0;

    int tile_idx = *moved_val - 1;
    sim->tiles[tile_idx].target.row = (float)sim->empty_row;
    sim->tiles[tile_idx].target.col = (float)sim->empty_col;
    sim->tiles[tile_idx].is_moving = true;

    sim->empty_row = (uint8_t)target_r;
    sim->empty_col = (uint8_t)target_c;
    sim->move_count++;

    return true;
}

static void update_physics(SimulationState *sim, float dt, EventQueue *events) {
    for (int i = 0; i < TOTAL_TILES - 1; ++i) {
        TileBody *tile = &sim->tiles[i];
        if (!tile->is_moving) continue;

        float dr = tile->target.row - tile->position.row;
        float dc = tile->target.col - tile->position.col;
        float dist = sqrtf(dr * dr + dc * dc);

        if (dist < 0.01f) {
            tile->position = tile->target;
            tile->velocity = (GridPosition){ 0.0f, 0.0f };
            tile->is_moving = false;

            if (events) {
                SimEvent e = {
                    .tick = sim->tick,
                    .type = EVT_TILE_SETTLED,
                    .tile_value = tile->value
                };
                event_queue_push(events, &e);
            }
        } else {
            float factor = 1.0f - expf(-TILE_SLIDE_SPEED * dt);
            tile->position.row += dr * factor;
            tile->position.col += dc * factor;
            tile->velocity.row = dr * TILE_SLIDE_SPEED;
            tile->velocity.col = dc * TILE_SLIDE_SPEED;
        }
    }
}

void sim_shutdown(SimulationState *state) {
    memset(state, 0, sizeof(*state));
}

void sim_begin_match(SimulationState *state, uint64_t seed, EventQueue *events) {
    memset(state, 0, sizeof(*state));
    state->seed = seed;
    rng_seed(&state->rng, seed, 1);

    uint8_t val = 1;
    for (int r = 0; r < BOARD_SIZE; ++r) {
        for (int c = 0; c < BOARD_SIZE; ++c) {
            state->logical_board[r][c] = val;
            if (val <= TOTAL_TILES - 1) {
                int idx = val - 1;
                state->tiles[idx].value = val;
                state->tiles[idx].position = (GridPosition){ (float)r, (float)c };
                state->tiles[idx].target   = (GridPosition){ (float)r, (float)c };
                state->tiles[idx].velocity = (GridPosition){ 0.0f, 0.0f };
                state->tiles[idx].is_moving = false;
            }
            val++;
        }
    }
    state->logical_board[BOARD_SIZE - 1][BOARD_SIZE - 1] = 0;
    state->empty_row = BOARD_SIZE - 1;
    state->empty_col = BOARD_SIZE - 1;

    // Guaranteed solvable shuffle via random valid moves, avoiding immediate reverse moves
    int moves_done = 0;
    int last_dir = -1;
    while (moves_done < 250) {
        int dir = (int)(rng_next_u64(&state->rng) % 4);
        // Avoid reversing previous move
        if ((dir == 0 && last_dir == 1) || (dir == 1 && last_dir == 0) ||
            (dir == 2 && last_dir == 3) || (dir == 3 && last_dir == 2)) {
            dir = (dir + 1) % 4;
        }

        int dr = 0, dc = 0;
        if (dir == 0) dr = -1;
        else if (dir == 1) dr = 1;
        else if (dir == 2) dc = -1;
        else if (dir == 3) dc = 1;

        int moved_val = 0;
        if (execute_slide(state, dr, dc, &moved_val)) {
            moves_done++;
            last_dir = dir;
            // FIX: Reset is_moving so shuffle isn't blocked by physical movement flags
            if (moved_val >= 1 && moved_val < TOTAL_TILES) {
                state->tiles[moved_val - 1].is_moving = false;
            }
        }
    }



    // Ensure the shuffle did not result in an already solved board
    if (check_solved(state)) {
        // Force one extra valid slide if solved
        int dr = (state->empty_row > 0) ? -1 : 1;
        int dummy = 0;
        execute_slide(state, dr, 0, &dummy);
    }

    // Snap physical positions to post-shuffle logical grid
    for (int r = 0; r < BOARD_SIZE; ++r) {
        for (int c = 0; c < BOARD_SIZE; ++c) {
            uint8_t v = state->logical_board[r][c];
            if (v != 0) {
                int idx = v - 1;
                state->tiles[idx].position = (GridPosition){ (float)r, (float)c };
                state->tiles[idx].target   = (GridPosition){ (float)r, (float)c };
                state->tiles[idx].is_moving = false;
            }
        }
    }

    state->move_count = 0;
    state->is_solved = false;
    state->match_over = false;

    if (events) {
        SimEvent e = { .tick = 0, .type = EVT_MATCH_STARTED };
        event_queue_push(events, &e);
    }
}

void sim_step(SimulationState *state, const TickInput *input, float dt, EventQueue *events) {
    if (state->match_over) return;

    state->tick++;

    // Only accept input if puzzle is not logically solved yet
    if (!state->is_solved) {
        int dr = 0, dc = 0;
        // Arrow keys direction convention: tile moves towards empty space
        if (input->move_dir == DIR_UP)    { dr = 1;  dc = 0; }
        if (input->move_dir == DIR_DOWN)  { dr = -1; dc = 0; }
        if (input->move_dir == DIR_LEFT)  { dr = 0;  dc = 1; }
        if (input->move_dir == DIR_RIGHT) { dr = 0;  dc = -1; }

        if (dr != 0 || dc != 0) {
            int moved_val = 0;
            if (execute_slide(state, dr, dc, &moved_val)) {
                if (events) {
                    SimEvent e = {
                        .tick = state->tick,
                        .type = EVT_TILE_SLID,
                        .tile_value = moved_val,
                        .score_player = (int)state->move_count
                    };
                    event_queue_push(events, &e);
                }

                if (check_solved(state)) {
                    state->is_solved = true;
                }
            }
        }
    }

    update_physics(state, dt, events);

    // Separating logical completion from physical completion
    if (state->is_solved && all_tiles_settled(state)) {
        state->match_over = true;
        if (events) {
            SimEvent win_event = {
                .tick = state->tick,
                .type = EVT_MATCH_ENDED,
                .score_player = (int)state->move_count
            };
            event_queue_push(events, &win_event);
        }
    }
}
