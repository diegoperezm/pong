#include "app.h"
#include "rng.h"
#include "simulation.h"
#include <stdarg.h>
#include <stdio.h>
#include <string.h>

#if defined(__GNUC__) || defined(__clang__)
__attribute__((format(printf, 2, 3)))
#endif
static void set_status(App *app, const char *fmt, ...) {
    va_list args;
    va_start(args, fmt);
    vsnprintf(app->status, sizeof app->status, fmt, args);
    va_end(args);
}

static void make_snapshot(const App *app, RenderSnapshot *snap) {
    const SimulationState *s = &app->sim;

    /* TODO: Copy game state values from s to snap for rendering/interpolation */
    snap->mode      = app->mode;
    snap->winner    = s->winner;
    snap->tick      = s->tick;
    snap->game_time = (float)s->tick * SIM_DT;
    (void)s;
}

static void snapshots_reset(App *app) {
    make_snapshot(app, &app->current);
    app->previous = app->current;
}

static void begin_match_common(App *app, uint64_t seed) {
    event_queue_clear(&app->events);
    fx_reset(&app->fx, seed, &app->events);
    sim_begin_match(&app->sim, seed, &app->events);
    fx_consume_events(&app->fx, &app->events);
    app->accumulator = 0.0;
}

static void start_live_match(App *app) {
    uint64_t seed = rng_splitmix64(app->base_seed + app->match_counter++);
    input_source_use_live(&app->source);
    begin_match_common(app, seed);

    app->mode = GAME_PLAYING;
    snapshots_reset(app);
    set_status(app, "match started (seed %016llx)", (unsigned long long)seed);
}

static void run_tick(App *app, bool snapshot) {
    SimulationState *sim = &app->sim;
    const uint32_t tick  = sim->tick;

    TickInput in = input_source_at(&app->source, tick);

    if (snapshot) app->previous = app->current;

    sim_step(sim, &in, (float)SIM_DT, &app->events);

    fx_consume_events(&app->fx, &app->events);
    fx_update(&app->fx, (float)SIM_DT);

    if (snapshot) make_snapshot(app, &app->current);

    if (sim->match_over) {
        app->mode = GAME_OVER;
        app->accumulator = 0.0;
    }
}

static void handle_input(App *app, const GameInput *in) {
    switch (app->mode) {
        case GAME_TITLE:
        case GAME_OVER:
            if (in->start) start_live_match(app);
            break;
        case GAME_PLAYING:
            if (in->pause) app->mode = GAME_PAUSED;
            break;
        case GAME_PAUSED:
            if (in->pause) app->mode = GAME_PLAYING;
            break;
    }
}

void app_init(App *app, uint64_t base_seed) {
    memset(app, 0, sizeof *app);
    app->mode      = GAME_TITLE;
    app->base_seed = base_seed;
    app->alpha     = 1.0f;

    input_source_use_live(&app->source);
    event_queue_clear(&app->events);
    fx_reset(&app->fx, 0, &app->events);
    snapshots_reset(app);
}

void app_shutdown(App *app) {
    sim_shutdown(&app->sim);
}

void app_frame(App *app, const GameInput *input, double frame_time) {
    GameMode before = app->mode;

    handle_input(app, input);

    if (app->mode != before) {
        app->accumulator = 0.0;
        snapshots_reset(app);
    }

    if (app->mode == GAME_PLAYING) {
        input_source_set_live_held(&app->source, (TickInput){ input->up, input->down });
        app->accumulator += frame_time;

        while (app->mode == GAME_PLAYING && app->accumulator >= (double)SIM_DT) {
            run_tick(app, true);
            app->accumulator -= (double)SIM_DT;
        }

        if (app->mode != GAME_PLAYING) {
            app->accumulator = 0.0;
            snapshots_reset(app);
            app->alpha = 1.0f;
        } else {
            app->alpha = (float)(app->accumulator / (double)SIM_DT);
        }
    } else {
        app->alpha = 1.0f;
        if (app->mode == GAME_PAUSED && input->step)
            run_tick(app, true);
    }
}

AppView app_view(const App *app) {
    return (AppView){ &app->previous, &app->current, app->alpha };
}

DebugHistory app_debug_history(const App *app) {
    DebugHistory h;
    memset(&h, 0, sizeof h);
    h.events      = &app->events;
    h.tick        = app->sim.tick;
    h.seed        = app->sim.seed;
    h.events_lost = app->fx.cursor.lost;
    h.status      = app->status;
    return h;
}

#include "debug.h"

#define DEBUG_FPS_SMOOTHING   0.10f  /* EMA weight of the newest sample */

void debug_init(DebugState *debug) {
    debug->enabled        = 0;
    debug->frame_time     = 0.0f;
    debug->fps            = 0.0f;
    debug->entity_count   = 0;
    debug->particle_count = 0;
}

void debug_update(DebugState *debug, const RenderSnapshot *current, float raw_frame_time) {
    if (debug == NULL || current == NULL || !debug->enabled) return;

    debug->frame_time = raw_frame_time;

    if (raw_frame_time > 0.0f) {
        float instant_fps = 1.0f / raw_frame_time;
        debug->fps = (debug->fps <= 0.0f)
            ? instant_fps
            : debug->fps + (instant_fps - debug->fps) * DEBUG_FPS_SMOOTHING;
    }
}

#if defined(__GNUC__) || defined(__clang__)
__attribute__((format(printf, 6, 7)))
#endif
static void text_line(int *x, int *y, int font, int line_height, Color color,
                      const char *fmt, ...) {
    char text[160];
    va_list args;
    va_start(args, fmt);
    vsnprintf(text, sizeof(text), fmt, args);
    va_end(args);

    DrawText(text, *x, *y, font, color);
    *y += line_height;
}

static const char *debug_game_mode_name(GameMode mode) {
    switch (mode) {
        case GAME_TITLE:   return "TITLE";
        case GAME_PLAYING: return "PLAYING";
        case GAME_PAUSED:  return "PAUSED";
        case GAME_OVER:    return "GAME OVER";
        default:           return "UNKNOWN";
    }
}

static void debug_draw_panel(const DebugState *debug, const RenderSnapshot *current) {
    DrawRectangle(DEBUG_PANEL_X, DEBUG_PANEL_Y, DEBUG_PANEL_WIDTH, DEBUG_PANEL_HEIGHT, Fade(BLACK, 0.80f));
    DrawRectangleLines(DEBUG_PANEL_X, DEBUG_PANEL_Y, DEBUG_PANEL_WIDTH, DEBUG_PANEL_HEIGHT, RAYWHITE);

    int x = DEBUG_PANEL_X + DEBUG_PADDING;
    int y = DEBUG_PANEL_Y + DEBUG_PADDING;
    const int F = DEBUG_FONT_SIZE, L = DEBUG_LINE_HEIGHT;

    text_line(&x, &y, F, L, RAYWHITE, "DEBUG");
    text_line(&x, &y, F, L, RAYWHITE, "FPS: %.1f", debug->fps);
    text_line(&x, &y, F, L, RAYWHITE, "Frame: %.3f ms", debug->frame_time * 1000.0f);
    text_line(&x, &y, F, L, RAYWHITE, "Mode: %s", debug_game_mode_name(current->mode));
    text_line(&x, &y, F, L, RAYWHITE, "Time: %.2f", current->game_time);
}

void debug_draw(
    const DebugState *debug,
    const RenderSnapshot *previous,
    const RenderSnapshot *current,
    float alpha,
    const DebugHistory *history)
{
    (void)previous;
    (void)alpha;
    (void)history;

    if (debug == NULL || current == NULL)
        return;
    if (!debug->enabled)
        return;

    debug_draw_panel(debug, current);
}

entities.c

#include "entities.h"
#include <string.h>
#include "log.h"

void entity_pool_init(EntityPool *pool) {
    memset(pool, 0, sizeof(*pool));

    pool->free_head   = 0;
    pool->next_serial = 1;

    for (uint32_t i = 0; i < MAX_ENTITIES; ++i) {
        pool->slots[i].next_free  = (i + 1 < MAX_ENTITIES) ? (i + 1) : INVALID_INDEX;
        pool->slots[i].generation = 1;
        pool->slots[i].active     = false;
        pool->slots[i].dense_idx  = INVALID_INDEX;
        pool->body[i]             = b2_nullBodyId;
        pool->dense_to_sparse[i]  = INVALID_INDEX;
    }
}

void entity_pool_clear(EntityPool *pool) {
    for (uint32_t i = 0; i < pool->count; ++i) {
        if (b2Body_IsValid(pool->body[i])) {
            b2DestroyBody(pool->body[i]);
        }
    }
    entity_pool_init(pool);
}

EntityHandle entity_create(EntityPool *pool, b2WorldId world, float x, float y) {
    if (!b2World_IsValid(world)) return INVALID_HANDLE;

    if (pool->free_head == INVALID_INDEX || pool->count >= MAX_ENTITIES) {
        LOG_ENTITY("entity creation failed: pool full");
        return INVALID_HANDLE;
    }

    uint32_t slot_idx = pool->free_head;
    Slot *slot = &pool->slots[slot_idx];
    pool->free_head = slot->next_free;

    uint32_t dense_idx = pool->count++;
    slot->dense_idx = dense_idx;
    slot->active = true;

    b2BodyDef body_def = b2DefaultBodyDef();
    body_def.type = b2_dynamicBody;
    body_def.position = pixel_to_meter(x, y);

    b2BodyId body = b2CreateBody(world, &body_def);
    if (!b2Body_IsValid(body)) {
        pool->count--;
        slot->active = false;
        slot->dense_idx = INVALID_INDEX;
        slot->next_free = pool->free_head;
        pool->free_head = slot_idx;
        return INVALID_HANDLE;
    }

    pool->body[dense_idx] = body;
    pool->x[dense_idx]    = x;
    pool->y[dense_idx]    = y;
    pool->dense_to_sparse[dense_idx] = slot_idx;
    pool->serial[dense_idx] = pool->next_serial++;

    return (EntityHandle){
        .index = slot_idx,
        .generation = slot->generation
    };
}

bool entity_is_valid(const EntityPool *pool, EntityHandle handle) {
    if (handle.index >= MAX_ENTITIES || handle.generation == 0) return false;

    const Slot *slot = &pool->slots[handle.index];
    if (!slot->active || slot->generation != handle.generation) return false;

    uint32_t dense_idx = slot->dense_idx;
    if (dense_idx >= pool->count) return false;

    return pool->dense_to_sparse[dense_idx] == handle.index;
}

void entity_destroy(EntityPool *pool, EntityHandle handle) {
    if (!entity_is_valid(pool, handle)) return;

    uint32_t slot_idx = handle.index;
    Slot *slot = &pool->slots[slot_idx];

    uint32_t dead_dense = slot->dense_idx;
    uint32_t last_dense = --pool->count;

    if (b2Body_IsValid(pool->body[dead_dense])) {
        b2DestroyBody(pool->body[dead_dense]);
    }

    if (dead_dense != last_dense) {
        pool->body[dead_dense]   = pool->body[last_dense];
        pool->x[dead_dense]      = pool->x[last_dense];
        pool->y[dead_dense]      = pool->y[last_dense];
        pool->serial[dead_dense] = pool->serial[last_dense];

        uint32_t moved_slot = pool->dense_to_sparse[last_dense];
        pool->dense_to_sparse[dead_dense] = moved_slot;
        pool->slots[moved_slot].dense_idx = dead_dense;
    }

    pool->body[last_dense]            = b2_nullBodyId;
    pool->x[last_dense]               = 0.0f;
    pool->y[last_dense]               = 0.0f;
    pool->serial[last_dense]          = 0;
    pool->dense_to_sparse[last_dense] = INVALID_INDEX;

    slot->active = false;
    slot->dense_idx = INVALID_INDEX;

    slot->generation++;
    if (slot->generation == 0) {
        slot->generation = 1;
    }

    slot->next_free = pool->free_head;
    pool->free_head = slot_idx;
}

uint32_t entities_sorted_by_serial(const EntityPool *pool, uint32_t order[MAX_ENTITIES]) {
    uint32_t n = pool->count;
    for (uint32_t i = 0; i < n; ++i) {
        uint32_t j = i;
        while (j > 0 && pool->serial[order[j - 1]] > pool->serial[i]) {
            order[j] = order[j - 1];
            --j;
        }
        order[j] = i;
    }
    return n;
}


events.h

#ifndef EVENTS_H
#define EVENTS_H

#include <stdbool.h>
#include <stdint.h>
#include "sim_types.h"

#define EVENT_RING_CAPACITY 256u   /* power of two */

_Static_assert((EVENT_RING_CAPACITY & (EVENT_RING_CAPACITY - 1u)) == 0u,
               "EVENT_RING_CAPACITY must be a power of two");

typedef enum {
    EVT_NONE = 0,
    EVT_MATCH_STARTED,      /* tick 0 */
    EVT_CUSTOM_EVENT_1,     /* Generic placeholder for game events */
    EVT_CUSTOM_EVENT_2,     /* Generic placeholder for game events */
    EVT_MATCH_ENDED         /* match / session over */
} EventType;

typedef struct {
    uint32_t  tick;
    EventType type;
    uint32_t  entity_serial;
    float     x, y;           /* coordinates / position */
    float     vx, vy;         /* velocities */
    int       param1;         /* generic custom parameters */
    int       param2;
} SimEvent;

/* Bounded history ring. Oldest entries are overwritten. */
typedef struct {
    SimEvent events[EVENT_RING_CAPACITY];
    uint64_t head;
} EventQueue;

/* Each consumer owns a cursor, so consumers never interfere. */
typedef struct {
    uint64_t next;   /* sequence number of the next event to read */
    uint32_t lost;   /* events overwritten before this consumer read them */
} EventCursor;

void     event_queue_clear(EventQueue *q);
void     event_queue_push(EventQueue *q, const SimEvent *e);   /* q may be NULL */
uint64_t event_queue_total(const EventQueue *q);

/* back = 0 is the newest event. Returns false if there is no such event. */
bool     event_queue_recent(const EventQueue *q, uint32_t back, SimEvent *out);

void     event_cursor_sync(const EventQueue *q, EventCursor *c);   /* skip to head */
bool     event_cursor_read(const EventQueue *q, EventCursor *c, SimEvent *out);

const char *event_type_name(EventType type);

#endif




events.c

#include "events.h"

#define RING_MASK (EVENT_RING_CAPACITY - 1u)

void event_queue_clear(EventQueue *q) {
    if (q) q->head = 0;
}

void event_queue_push(EventQueue *q, const SimEvent *e) {
    if (!q || !e) return;
    q->events[q->head & RING_MASK] = *e;
    q->head++;
}

uint64_t event_queue_total(const EventQueue *q) {
    return q ? q->head : 0;
}

bool event_queue_recent(const EventQueue *q, uint32_t back, SimEvent *out) {
    if (!q || !out) return false;
    uint64_t available = q->head < EVENT_RING_CAPACITY ? q->head : EVENT_RING_CAPACITY;
    if ((uint64_t)back >= available) return false;
    *out = q->events[(q->head - 1u - back) & RING_MASK];
    return true;
}

void event_cursor_sync(const EventQueue *q, EventCursor *c) {
    c->next = q ? q->head : 0;
}

bool event_cursor_read(const EventQueue *q, EventCursor *c, SimEvent *out) {
    if (!q || !c || !out) return false;

    if (c->next > q->head)            /* queue was cleared under us */
        c->next = q->head;

    uint64_t oldest = q->head > EVENT_RING_CAPACITY ? q->head - EVENT_RING_CAPACITY : 0;
    if (c->next < oldest) {           /* fell behind: those events are gone */
        c->lost += (uint32_t)(oldest - c->next);
        c->next = oldest;
    }

    if (c->next >= q->head) return false;

    *out = q->events[c->next & RING_MASK];
    c->next++;
    return true;
}

const char *event_type_name(EventType type) {
    switch (type) {
        case EVT_MATCH_STARTED:    return "MATCH_START";
        case EVT_CUSTOM_EVENT_1:   return "CUSTOM_1";
        case EVT_CUSTOM_EVENT_2:   return "CUSTOM_2";
        case EVT_MATCH_ENDED:      return "MATCH_END";
        default:                   return "NONE";
    }
}

#include <string.h>

static void particles_clear(ParticlePool *particles) {
    particles->free_head = 0;

    for (int i = 0; i < MAX_PARTICLES; ++i) {
        particles->active[i] = 0;
        particles->next_free[i] = (i + 1 < MAX_PARTICLES) ? (i + 1) : -1;
    }
}

static void particles_update(ParticlePool *particles, float dt) {
    /* Frame-rate independent damping (normalised to 60 FPS). */
    float damping = powf(0.98f, dt * 60.0f);

    for (int i = 0; i < MAX_PARTICLES; ++i) {
        if (!particles->active[i]) continue;

        particles->lifetime[i] -= dt;
        if (particles->lifetime[i] <= 0.0f) {
            particles->active[i] = 0;
            particles->next_free[i] = particles->free_head;
            particles->free_head    = i;
            continue;
        }

        particles->x[i]  += particles->vx[i] * dt;
        particles->y[i]  += particles->vy[i] * dt;
        particles->vx[i] *= damping;
        particles->vy[i] *= damping;
    }
}

void fx_reset(Fx *fx, uint64_t seed, const EventQueue *queue) {
    memset(fx, 0, sizeof *fx);
    particles_clear(&fx->particles);
    rng_seed(&fx->rng, seed, 2);
    event_cursor_sync(queue, &fx->cursor);
}

void input_sample(GameInput *input) {
    /* Continuous movement checks (support both Arrows and WASD) */
    input->up    = IsKeyDown(KEY_UP) || IsKeyDown(KEY_W);
    input->down  = IsKeyDown(KEY_DOWN) || IsKeyDown(KEY_S);
    input->left  = IsKeyDown(KEY_LEFT) || IsKeyDown(KEY_A);
    input->right = IsKeyDown(KEY_RIGHT) || IsKeyDown(KEY_D);

    /* Edge-triggered action keys */
    input->action1 = IsKeyPressed(KEY_J) || IsKeyPressed(KEY_Z);
    input->action2 = IsKeyPressed(KEY_K) || IsKeyPressed(KEY_X);

    /* System controls (edge-triggered: true for exactly one frame) */
    input->start        = IsKeyPressed(KEY_ENTER) || IsKeyPressed(KEY_SPACE);
    input->pause        = IsKeyPressed(KEY_P) || IsKeyPressed(KEY_ESCAPE);
    input->step         = IsKeyPressed(KEY_N); /* Useful for stepping frames when paused */
    input->debug_toggle = IsKeyPressed(KEY_F3);
}

#include "log.h"
#include <stdarg.h>
#include <stdio.h>

static uint32_t enabled_categories;

#define LOG_CATEGORY_NAME(category, name) [category] = name,

const char* log_category_name[LOG_COUNT] = {
    LOG_CATEGORIES(LOG_CATEGORY_NAME)
};

#undef LOG_CATEGORY_NAME

void log_init(void) {
    enabled_categories = 0;
}

void log_enable(LogCategory category) {
    if (category >= LOG_COUNT || category >= 32) return;
    enabled_categories |= (1u << category);
}

void log_disable(LogCategory category) {
    if (category >= LOG_COUNT || category >= 32) return;
    enabled_categories &= ~(1u << category);
}

int log_is_enabled(LogCategory category) {
    return (enabled_categories & (1u << category)) != 0; 
}

void log_message(
    LogCategory category,
    const char* file,
    int         line,
    const char* format,
    ...
) {
    if (!log_is_enabled(category))
        return;

    printf(
        "[%s] %s:%d: ",
        log_category_name[category],
        file,
        line
    );

    va_list args;
    va_start(args, format);
    vprintf(format, args);
    va_end(args);

    printf("\n");
    fflush(stdout);
}

/* main() is only the platform shell: window, input sampling, drawing.
 * Everything about the game itself lives in the app and simulation layers. */

int main(void) {
    InitWindow(
        SCREEN_WIDTH,
        SCREEN_HEIGHT,
        "Game Template"
    );

    /* ESC is the pause key; without this raylib would close the window. */
    SetExitKey(KEY_NULL);
    SetTargetFPS(60);

    /* Logging must be configured BEFORE anything that logs. */
    log_init();

    /* ~100 KB (two snapshots): keep it off the stack. Zero-initialised. */
    static App app;
    app_init(&app, (uint64_t)time(NULL));

    DebugState debug;
    debug_init(&debug);
    debug.enabled = 1;

    while (!WindowShouldClose()) {
        GameInput input;
        input_sample(&input);

        /* Real duration of the last frame: debug readout only. */
        double raw_frame_time = GetFrameTime();

        /* Clamped copy: drives the simulation (spiral-of-death guard). */
        double frame_time = raw_frame_time;
        if (frame_time > MAX_FRAME_TIME)
            frame_time = MAX_FRAME_TIME;

        if (input.debug_toggle)
            debug.enabled = !debug.enabled;

        app_frame(&app, &input, frame_time);

        AppView view = app_view(&app);
        DebugHistory history = app_debug_history(&app);

        debug_update(&debug, view.current, (float)raw_frame_time);

        BeginDrawing();
            ClearBackground(BLACK);

            render_frame(view.previous, view.current, view.alpha);
            debug_draw(&debug, view.previous, view.current, view.alpha, &history);

        EndDrawing();
    }

    app_shutdown(&app);
    CloseWindow();

    return 0;
}

#include "rng.h"

uint64_t rng_splitmix64(uint64_t state) {
    state += 0x9e3779b97f4a7c15ULL;
    state = (state ^ (state >> 30)) * 0xbf58476d1ce4e5b9ULL;
    state = (state ^ (state >> 27)) * 0x94d049bb133111ebULL;
    return state ^ (state >> 31);
}

void rng_seed(Rng *rng, uint64_t seed, uint64_t stream) {
    rng->state = 0U;
    rng->inc = (stream << 1U) | 1U;
    rng_next_u32(rng);
    rng->state += seed;
    rng_next_u32(rng);
}

uint32_t rng_next_u32(Rng *rng) {
    uint64_t oldstate = rng->state;
    rng->state = oldstate * 6364136223846793005ULL + rng->inc;
    uint32_t xorshifted = (uint32_t)(((oldstate >> 18U) ^ oldstate) >> 27U);
    uint32_t rot = (uint32_t)(oldstate >> 59U);
    return (xorshifted >> rot) | (xorshifted << ((-rot) & 31));
}

int rng_range(Rng *rng, int min, int max) {
    if (min >= max) return min;
    uint32_t range = (uint32_t)(max - min + 1);
    return min + (int)(rng_next_u32(rng) % range);
}

float rng_float_range(Rng *rng, float min, float max) {
    uint32_t val = rng_next_u32(rng);
    float norm = (float)val / (float)UINT32_MAX;
    return min + norm * (max - min);
}
